# av-to-transcript-and-minutes 依赖声明
#
# 安装（建议在虚拟环境或 conda 环境中）：
#   pip install -r requirements.txt
# GPU 加速需先按 https://pytorch.org 指引安装对应 CUDA 版本的 torch。

# --- 语音转写（核心）---
torch>=2.0                # 需 CUDA 版以获得 GPU 加速（CPU 亦可运行，速度慢）
torchaudio                # 音频读写（与 torch 版本配套）
transformers>=4.45        # Qwen3-ASR 模型加载（需支持 Qwen3ASRForConditionalGeneration）
funasr>=1.1               # VAD 切段 / cam++ 声纹 / ct-punc 标点 / fa-zh 强制对齐
librosa                   # 音频重采样
soundfile                 # wav 读写
numpy
scikit-learn              # 声纹层次聚类（cam++ 引擎）
opencc-python-reimplemented  # 繁→简转换（默认启用）

# --- 人声/背景音分离 ---
demucs>=4.0               # htdemucs 两轨分离（--no-separate 可跳过）

# --- 说话人分离（pyannote 引擎，会议场景推荐）---
pyannote.audio>=4.0       # community-1 分割模型（重叠语音更准）
# pyannote.metrics 随 pyannote.audio 装上，供 benchmark_quality.py 算 DER（缺则只算 CER）。
# 注意：pyannote.audio 会引入 torchcodec，而 torchcodec 要求 FFmpeg full-shared 版；
# 若报 libtorchcodec 加载失败，执行：pip uninstall torchcodec
# 本项目用 soundfile 预载波形，不依赖 torchcodec。

# --- 可选：多语言字幕对齐（--align-engine native）---
# 需另行下载 Qwen/Qwen3-ForcedAligner-0.6B（约 1.2GB）：
#   python -c "from modelscope import snapshot_download; snapshot_download('Qwen/Qwen3-ForcedAligner-0.6B')"
# 未下载时字幕自动回落中文专用的 fa-zh；非中文素材必须装它，否则时间戳错位。

# --- 可选：更完整的 ITN 规则库（--itn-backend wtp）---
# WeTextProcessing 依赖 pynini，而 pynini 在 PyPI 只有源码包，Windows 下需 MSVC 编译。
# 装不上不影响使用：ITN 回落自写零依赖规则（覆盖窄一些）。
# WeTextProcessing

# --- 文稿排版（article-format 技能，按需）---
python-docx               # 生成规范 docx
lxml

# --- 可选：FireRedASR2-AED 备选识别引擎（中/英/粤；v2.1 起长音频显存与速度已优化）---
# 见 qwen_asr.py 头部注释中的模型与代码获取方式
# pypinyin rapidfuzz      # 专名纠错拼音模糊匹配（--fuzzy）

# --- 可选：云端 ASR 引擎（--asr-engine cloud，v2.3）---
# dashscope               # 阿里云百炼 SDK（message 通道需要；omni/filetrans 通道走纯 HTTP 不需要）
# 另需环境变量 DASHSCOPE_API_KEY（阿里云百炼控制台获取）

# --- 系统依赖（非 pip）---
# FFmpeg（含 ffmpeg / ffprobe，需在 PATH 中）——拆轨、噪声门、字幕合成
#   Windows: winget install ffmpeg  或 https://www.gyan.dev/ffmpeg/builds/
#   Linux:   apt install ffmpeg
#   macOS:   brew install ffmpeg
